CString CSRadView::TipoPan(int ttt)
{
	if (ttt>8 || ttt<1)
		return "";

	switch (ttt)
	{
		case 1:
			return "Proter QF";
		case 2:
			return "Proter QM";
		case 3:
			return "Proter FLY";
		case 4:
			return "Proter alfa-PARALLEL";
		case 5:
			return "Proter alfa-CROSS";
		case 6:
			return "Doghe passo 200 mm";
		case 7:
			return "Proter GKM";
		case 8:
			return "Proter GP";
	}
}


BOOL Dialog5::OnInitDialog() 
{
	CDialog::OnInitDialog();
	
	switch (tp)
	{
		case 1:  //  QF
		{
		ShowArt("gruppo1","art1_1");	
				m_CatForm.SetValore("xxx","200");
			if (dpx==600)
				m_CatForm.SetValore("dim1_1","2");
			else
			{
				if (dpy==600)
					m_CatForm.SetValore("dim1_1","0");
				else
					m_CatForm.SetValore("dim1_1","1");
			}
		break;
		}
		case 2:  //  QM
		{
		ShowArt("gruppo1","art1_2");	
		m_CatForm.SetValore("sc1_2",Str(ts-3,0));
		break;
		}
		case 3:  //  Fly
		{
		ShowArt("gruppo2","art2");	
		m_CatForm.SetValore("h2",Str(dpx,0));
		m_CatForm.SetValore("l2",Str(dpy,0));
		m_CatForm.SetValore("sc2",Str(ts-3,0));
		break;
		}
		case 4:  //  alfa-parallel
		{
		ShowArt("gruppo3","art3_1");	
		m_CatForm.SetValore("h3_1",Str(dpx,0));
		m_CatForm.SetValore("l3_1",Str(dpy,0));
		m_CatForm.SetValore("sc3_1",Str(ts-3,0));
		break;
		}
		case 5:  //  alfa-cross
		{
		ShowArt("gruppo3","art3_2");	
		m_CatForm.SetValore("h3_2",Str(dpx,0));
		m_CatForm.SetValore("l3_2",Str(dpy,0));
		m_CatForm.SetValore("sc3_2",Str(ts-3,0));
		break;
		}
		case 6:  //  doghe
		{
		ShowArt("gruppo4","art4");	
		m_CatForm.SetValore("l4",Str(dpy,0));
		m_CatForm.SetValore("sc4",Str(ts-3,0));
		break;
		}
		case 7:  //  GKM
		{
		ShowArt("gruppo5","art5_1");	
		break;
		}
		case 8:  //  GP
		{
		ShowArt("gruppo5","art5_2");	
		break;
		}
	}
		
	m_CatForm.Refr();

	return TRUE;  // return TRUE unless you set the focus to a control
	              // EXCEPTION: OCX Property Pages should return FALSE
}

void Dialog5::OnOK() 
{
	

	CString art=m_CatForm.GetArticolo ();
	int vi;
	
	m_CatForm.Uscita();

	if (art=="art1_1")
	{
		tp=1;
		vi=atof(m_CatForm.GetValore ("dim1_1"));
		if (vi<2)
			dpx=300;
		else
			dpx=600;
		if (vi==1)
			dpy=1200;
		else
			dpy=600;
	}
	if (art=="art1_2")
	{
		tp=2;
		vi=atof(m_CatForm.GetValore("dim1_2"));
		if (vi==0)
		{
			dpx=300;
			dpy=1200;
		}
		else
		{
			dpx=600;
			dpy=600;
		}
	ts=atof(m_CatForm.GetValore("sc1_2"))+3;
	}
	if (art=="art2")
	{
		tp=3;
		dpx=atof(m_CatForm.GetValore("h2"));
		dpy=atof(m_CatForm.GetValore("l2"));
		ts=atof(m_CatForm.GetValore("sc2"))+3;
	}
	if (art=="art3_1")
	{
		tp=4;
		dpx=atof(m_CatForm.GetValore("h3_1"));
		dpy=atof(m_CatForm.GetValore("l3_1"));
		ts=atof(m_CatForm.GetValore("sc3_1"))+3;
	}
	if (art=="art3_2")
	{
		tp=5;
		dpx=atof(m_CatForm.GetValore("h3_2"));
		dpy=atof(m_CatForm.GetValore("l3_2"));
		ts=atof(m_CatForm.GetValore("sc3_2"))+3;
	}
	if (art=="art4")
	{
		tp=6;
		dpx=200;
		dpy=atof(m_CatForm.GetValore("l4"));
		ts=atof(m_CatForm.GetValore("sc4"))+3;
	}
	if (art=="art5_1")
		tp=7;
	if (art=="art5_2")
		tp=8;
	
	CDialog::OnOK();
}

void Dialog5::ShowArt(CString gruppo, CString art)
{
	m_CatForm.SetArticolo (gruppo,art);	
	m_CatForm.Show();

}
