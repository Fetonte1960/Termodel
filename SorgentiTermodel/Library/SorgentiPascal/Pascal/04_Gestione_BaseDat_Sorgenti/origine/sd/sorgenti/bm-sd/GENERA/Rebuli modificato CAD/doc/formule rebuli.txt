double ResaI(int p, double dt, int ps, int sc)
{
	switch (p)
	{
		case 1:  //  QF
		{
			switch (ps)
			{
				case 300:
					return 5.8*pow(dt,1.072);
				case 600:
					return 3.6*pow(dt,1.021);
			}
		break;
		}
		case 2:  // QM
		{
			return EmI(sc,dt);
		break;
		}
		case 3:  // FLY
		{

			return EmI(sc,dt);
		break;
		}
		case 4:  // alfa-parallel
		{
			return EmI(sc,dt);
		break;
		}
		case 5:  // alfa-cross
		{
			return EmI(sc,dt);
		break;
		}
		case 6:  // doghe
		{
			return EmI(sc,dt);
		break;
		}
		case 7:  // Cartongesso GKM
		{
			return 3.7*pow(dt,1.024);
		break;
		}
		case 8:  // Cartongesso GP
		{
			return 3.11*pow(dt,1.037);
		break;
		}
	}
}

double ResaE(int p, double dt, int ps, int sc)
{
	
	switch (p)
	{
		case 1:  //  QF
		{
			switch (ps)
			{
				case 300:
					return 4.9*pow(dt,1.037);
				case 600:
					return 2.52*pow(dt,1.016);
			}
		}
		break;
		case 2:  // QM
		{
			return EmE(sc,dt);
		break;
		}
		case 3:  // FLY
		{
			return EmE(sc,dt);
		break;
		}
		case 4:  // alfa-parallel
		{
			return EmE(sc,dt);
		break;
		}
		case 5:  // alfa-cross
		{
			return EmE(sc,dt);
		break;
		}
		case 6:  // doghe
		{
			return EmE(sc,dt);
		break;
		}
		case 7:  // Cartongesso GKM
		{
			return 4.96*pow(dt,1.096);
		break;
		}
		case 8:  // Cartongesso GP
		{
			return 4.65*pow(dt,1.082);
		break;
		}
	}
}
	double EmI(int n, double dt)	
{
	// Resa inv. degli scambiatori

		switch (n)
		{
		case 21: // 2T105
			return 6.03*pow(dt,1.078);
		case 22: // 2F105
			return 4.73*pow(dt,1.039);
		case 3: // 3T70
			return 6.03*pow(dt,1.078);
		case 4: // 4T70
			return 6.17*pow(dt,1.116);
		}
}
	double EmE(int n, double dt)	
{
	// Resa est. degli scambiatori

		switch (n)
		{
		case 21: // 2T105
			return 6.453*pow(dt,1.064);
		case 22: // 2F105
			return 5.65*pow(dt,1.071);
		case 3: // 3T70
			return 6.453*pow(dt,1.064);
		case 4: // 4T70
			return 6.751*pow(dt,1.163);
		}
}
